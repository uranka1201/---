Spreadsheet дээр Extensions → Apps Script нээнэ үү.
Code.gs-ийн бүх агуулгыг устгаад доорх кодыг оруулж Хадгалах, дараа нь installTrigger функцийг Run хийнэ үү.

// ============================================================
// БАГЦ ДҮРМИЙН ТЕСТ — Google Apps Script
// Google Form submission → Google Sheets бүртгэл
// ============================================================

const SHEET_NAME = 'Тайлан';
const FORM_SHEET = 'Form Responses 1'; // Form хариултын sheet нэр

// ------------------------------------------------------------
// Form submit үед автоматаар дуудагдана
// ------------------------------------------------------------
function onFormSubmit(e) {
  try {
    const ss   = SpreadsheetApp.getActiveSpreadsheet();
    let   sheet = ss.getSheetByName(SHEET_NAME);

    // Тайлан sheet байхгүй бол үүсгэнэ
    if (!sheet) {
      sheet = ss.insertSheet(SHEET_NAME);
      const headers = ['Огноо','Овог нэр','Албан тушаал','Байршил',
                       'Горим','Оноо %','Зөв','Нийт','Үр дүн','Хугацаа'];
      sheet.appendRow(headers);
      const hRange = sheet.getRange(1, 1, 1, headers.length);
      hRange.setFontWeight('bold')
            .setBackground('#1a56db')
            .setFontColor('white')
            .setHorizontalAlignment('center');
      sheet.setFrozenRows(1);
    }

    // Form хариултаас утга авна
    // Асуултын дараалал: Нэр, Тушаал, Байршил, Горим, Оноо, Зөв, Нийт, Үр дүн, Хугацаа
    const resp   = e.namedValues;
    const date   = new Date().toLocaleString('mn-MN', {timeZone: 'Asia/Ulaanbaatar'});
    const name   = (resp['Овог нэр']    || [''])[0];
    const tushaal= (resp['Албан тушаал']|| [''])[0];
    const gaazar = (resp['Байршил']     || [''])[0];
    const mode   = (resp['Горим']       || [''])[0];
    const score  = Number((resp['Оноо %']|| ['0'])[0]);
    const correct= Number((resp['Зөв']  || ['0'])[0]);
    const total  = Number((resp['Нийт'] || ['0'])[0]);
    const passed = score >= 60 ? 'ТЭНЦСЭН' : 'ТЭНЦЭЭГҮЙ';
    const time   = (resp['Хугацаа']     || [''])[0];

    sheet.appendRow([date, name, tushaal, gaazar, mode, score, correct, total, passed, time]);

    // Үр дүнгийн нүдэнд өнгө өгнө
    const lastRow = sheet.getLastRow();
    const resultCell = sheet.getRange(lastRow, 9);
    if (passed === 'ТЭНЦСЭН') {
      resultCell.setBackground('#d1fae5').setFontColor('#057a55');
    } else {
      resultCell.setBackground('#fee2e2').setFontColor('#e02424');
    }

    // Оноо баганад өнгөт мөр
    const scoreCell = sheet.getRange(lastRow, 6);
    scoreCell.setFontWeight('bold');
    if (score >= 80)      scoreCell.setFontColor('#057a55');
    else if (score >= 60) scoreCell.setFontColor('#0369a1');
    else                  scoreCell.setFontColor('#e02424');

  } catch (err) {
    console.error('onFormSubmit error:', err);
  }
}

// ------------------------------------------------------------
// Trigger суулгах — нэг удаа ажиллуулна
// ------------------------------------------------------------
function installTrigger() {
  // Өмнөх trigger-уудыг устгана
  ScriptApp.getProjectTriggers().forEach(t => ScriptApp.deleteTrigger(t));

  // Form submit trigger
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  ScriptApp.newTrigger('onFormSubmit')
    .forSpreadsheet(ss)
    .onFormSubmit()
    .create();

  SpreadsheetApp.getUi().alert('Trigger амжилттай суулгагдлаа!');
}

// ------------------------------------------------------------
// Тайлан sheet-г цэвэрлэх (толгой мөрийг үлдээнэ)
// ------------------------------------------------------------
function clearReport() {
  const ss    = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName(SHEET_NAME);
  if (!sheet || sheet.getLastRow() < 2) return;
  sheet.deleteRows(2, sheet.getLastRow() - 1);
}

// ------------------------------------------------------------
// Нэгдсэн статистик тооцох
// ------------------------------------------------------------
function buildSummary() {
  const ss     = SpreadsheetApp.getActiveSpreadsheet();
  const sheet  = ss.getSheetByName(SHEET_NAME);
  if (!sheet || sheet.getLastRow() < 2) {
    SpreadsheetApp.getUi().alert('Өгөгдөл байхгүй байна.');
    return;
  }

  let sumSheet = ss.getSheetByName('Дүн');
  if (sumSheet) ss.deleteSheet(sumSheet);
  sumSheet = ss.insertSheet('Дүн');

  const rows   = sheet.getRange(2, 1, sheet.getLastRow()-1, 10).getValues();
  const byRole = {};

  rows.forEach(r => {
    const role   = r[4] || 'бусад';
    const score  = Number(r[5]) || 0;
    const passed = r[8] === 'ТЭНЦСЭН';
    if (!byRole[role]) byRole[role] = {count:0, passed:0, totalScore:0};
    byRole[role].count++;
    if (passed) byRole[role].passed++;
    byRole[role].totalScore += score;
  });

  sumSheet.appendRow(['Горим','Нийт','Тэнцсэн','Тэнцээгүй','Дундаж оноо %','Тэнцсэн хувь %']);
  const hRange = sumSheet.getRange(1,1,1,6);
  hRange.setFontWeight('bold').setBackground('#1a56db').setFontColor('white');

  Object.entries(byRole).forEach(([role, d]) => {
    const avg     = Math.round(d.totalScore / d.count);
    const passPct = Math.round((d.passed / d.count) * 100);
    sumSheet.appendRow([role, d.count, d.passed, d.count - d.passed, avg, passPct]);
  });

  sumSheet.autoResizeColumns(1, 6);
  SpreadsheetApp.getUi().alert('Дүн sheet үүсгэгдлаа!');
}
